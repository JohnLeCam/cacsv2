⚠️ La table « transcripts » n'existe pas encore dans Supabase
Supabase → SQL Editor → New query → colle ce code → Run, puis recharge cette page.
create table if not exists public.transcripts ( id uuid primary key default gen_random_uuid(), channel_id text not null, channel_name text, client_discord_id text, client_username text, closed_by text, message_count integer default 0, html text not null, created_at timestamptz not null default now(), number bigint generated by default as identity ); create index if not exists transcripts_channel_id_idx on public.transcripts (channel_id); alter table public.transcripts enable row level security;
